S = "prog" "FIO" ";" Block "eof".
Block = DeclarationSection OperatorSection.
DeclarationSection = VarSection | TypeSection VarSection.
TypeSection = "using" TypeList.
TypeList = TypeDefinition ";" | TypeList TypeDefinition ";".
VarSection = "var" VarList.
VarList = VarDefinition ";" | VarList VarDefinition ";".
TypeDefinition = "id" CheckRedeclaration Type {DeclareType}.
VarDefinition = "id" CheckRedeclaration ":" Type {DeclareVariable}.
Type = "id" | StructureType.
StructureType = "class" "{" StructureFields "}".
StructureFields  = "id" ":" Type ";" | StructureFields "id" ":" Type ";".
OperatorSection = "{" OperatorList "}".
OperatorList = Operator ";" | OperatorList Operator ";".
Operator = AssignOperator | ControlOperator.
AssignOperator = Var "ass" Expr.
Var = "id" | Var "." "id".
ControlOperator = "while" "(" Expr ")" "{" OperatorList "}".
Expr = SimpleExpr | SimpleExpr "rel" SimpleExpr.
SimpleExpr = Term | SimpleExpr "+" Term | SimpleExpr "||" Term.
Term = Factor | Term "*" Factor | Term "&amp;&amp;" Factor.
Factor = Const | Var | "(" Expr ")" | "!" Factor.
Const = "num" | "true" | "false".
CheckRedeclaration = {CheckRedeclaration}.
!
