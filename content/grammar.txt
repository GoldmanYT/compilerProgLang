S = "prog" "FIO" ";" Block "eof".
Block = SecDecl SecOper.
SecDecl = SecVar | SecType SecVar.
SecType = "using" LstType.
LstType = DefType ";" | LstType DefType ";".
SecVar = "var" LstVar.
LstVar = DefVar ";" | LstVar DefVar ";".
DefType = "id" Type.
DefVar = "id" ":" Type.
Type = "id" | TypeStruct.
TypeStruct = "class" "{" FieldsStruct "}".
FieldsStruct  = "id" ":" Type ";" | FieldsStruct "id" ":" Type ";".
SecOper = "{" LstOper "}".
LstOper = Oper ";" | LstOper Oper ";".
Oper = OperAss | OperCtrl.
OperAss = Var "ass" Expr.
Var = "id" | Var "." "id".
OperCtrl = "while" "(" Expr ")" "{" LstOper "}".
Expr = SimpleExpr | SimpleExpr "rel" SimpleExpr.
SimpleExpr = Term | SimpleExpr "+" Term | SimpleExpr "||" Term.
Term = Factor | Term "*" Factor | Term "&&" Factor.
Factor = Const | Var | "(" Expr ")" | "!" Factor.
Const = "num" | "true" | "false".!
